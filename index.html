    // Исходные данные (со структурой "Человек -> Культовые места")

    // Инициализация карты

    const map = L.map('map').setView([55.0, 50.0], 5);


    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {

      maxZoom: 18,

      attribution: '© OpenStreetMap'

    }).addTo(map);


    let activeMarkers = [];


    // Рендер вкладок людей

    const tabsContainer = document.getElementById('personTabs');

    const detailsContainer = document.getElementById('personDetails');


    function renderTabs() {

      tabsContainer.innerHTML = '';

      familyData.forEach((person, index) => {

        const btn = document.createElement('button');

        btn.className = `person-tab-btn ${index === 0 ? 'active' : ''}`;

        btn.textContent = person.name.split(' ')[0] + ' ' + person.name.split(' ')[1][0] + '.';

        btn.onclick = () => selectPerson(person.id);

        tabsContainer.appendChild(btn);

      });

    }


    // Выбор персоны и обновление карты

    function selectPerson(personId) {

      document.querySelectorAll('.person-tab-btn').forEach((btn, idx) => {

        btn.classList.toggle('active', familyData[idx].id === personId);

      });


      const person = familyData.find(p => p.id === personId);

      if (!person) return;


      // Обновляем карточку персоны слева

      let placesHtml = person.places.map((place, idx) => `

        <div class="place-card" onclick="zoomToPlace(${idx})">

          <div class="place-type">${place.type} • ${place.city}</div>

          <div class="place-name">${place.title}</div>

          <div class="place-years">${place.years}</div>

          <div class="place-desc">${place.description}</div>

        </div>

      `).join('');


      detailsContainer.innerHTML = `

        <div class="person-header">

          <h2>${person.name}</h2>

          <div class="years">${person.years} (${person.relation})</div>

        </div>

        <div class="places-section">

          <h3>

            Культовые места 

            <button class="btn-add-place" onclick="alert('Форма добавления точки появится на следующем шаге')">+ Точка</button>

          </h3>

          ${placesHtml}

        </div>

      `;


      // Чистим старые маркеры и добавляем новые

      activeMarkers.forEach(m => map.removeLayer(m));

      activeMarkers = [];


      const bounds = [];

      person.places.forEach((place) => {

        const marker = L.circleMarker([place.lat, place.lng], {

          radius: 8,

          fillColor: '#C87D55',

          color: '#2D4436',

          weight: 2,

          opacity: 1,

          fillOpacity: 0.9

        }).addTo(map);


        marker.bindPopup(`<b>${place.title}</b><br>${place.type}<br><i>${place.years}</i>`);

        activeMarkers.push(marker);

        bounds.push([place.lat, place.lng]);

      });


      if (bounds.length > 0) {

        map.fitBounds(bounds, { padding: [50, 50], maxZoom: 9 });

      }

    }


    // Приближение карты при клике на конкретное место

    window.zoomToPlace = function(index) {

      const marker = activeMarkers[index];

      if (marker) {

        map.setView(marker.getLatLng(), 11, { animate: true });

        marker.openPopup();

      }

    };


    // Запуск

    renderTabs();

    selectPerson(familyData[0].id);
